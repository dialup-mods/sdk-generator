cmake_minimum_required(VERSION 4.0)

set(DIALUP_INSTALL_DIR "$ENV{LOCALAPPDATA}/DialUp")
include("${DIALUP_INSTALL_DIR}/build-tools/PreProject.cmake")

project(DialUp-SDKGen VERSION 1.3 LANGUAGES CXX)

option(DEPLOY_TO_DEV "Copy plugin and PDB to dev plugins folder after build" OFF)

include("${DIALUP_INSTALL_DIR}/build-tools/GameBinaryDir.cmake")

set(GAME "rocketleague" CACHE STRING "Name of the game config to compile with")
set(LOG_DIR "${CMAKE_SOURCE_DIR}/log" CACHE STRING "Directory to create the log file in")
file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}/config" CONFIG_DIR)

# fixme: make work with Makefile
file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}/plugin/generated" SDK_OUTPUT_DIR)

file(TO_CMAKE_PATH "${CONFIG_DIR}/${GAME}" GAME_CONFIG_DIR)
file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}/include/runtime" ENGINE_RUNTIME_DIR)
file(TO_CMAKE_PATH "${CMAKE_SOURCE_DIR}/include/model" UE_MODEL_DIR)
file(TO_CMAKE_PATH "${LOG_DIR}" LOG_DIR)
file(TO_CMAKE_PATH "${DIALUP_INSTALL_DIR}" DIALUP_DIR)

# output dirs
set(GENERATED_DIRS
    "${CMAKE_SOURCE_DIR}/config/rocketleague"
    "${CMAKE_SOURCE_DIR}/plugin/generated"
    "${CMAKE_SOURCE_DIR}/plugin/generated/impl"
    "${CMAKE_SOURCE_DIR}/plugin/generated/include"
    "${CMAKE_SOURCE_DIR}/plugin/imported"
    "${CMAKE_SOURCE_DIR}/plugin/imported/model"
    "${CMAKE_SOURCE_DIR}/plugin/imported/model/abstract"
    "${CMAKE_SOURCE_DIR}/plugin/imported/model/base"
    "${CMAKE_SOURCE_DIR}/plugin/imported/model/helpers"
    "${CMAKE_SOURCE_DIR}/plugin/imported/model/properties"
    "${CMAKE_SOURCE_DIR}/plugin/imported/runtime"
)
foreach(dir IN LISTS GENERATED_DIRS)
    file(MAKE_DIRECTORY "${dir}")

    set(warning_file "${dir}/!! WARNING_VOLATILE_DIRECTORY.md")
    if(NOT EXISTS "${warning_file}")
        file(WRITE "${warning_file}"
                "### ⚠️  GENERATED DIRECTORY

This directory is generated by the Dial-Up build system.

- Contents may be overwritten or deleted
- Do NOT edit files here
- Source of truth lives elsewhere
")
    endif()
endforeach()


message("Using game: ${GAME}")
message("Using schema from: ${CONFIG_DIR}/${GAME}")

# for AST walking Schema
set(CLANG_INCLUDE_DIR "$ENV{ProgramFiles}/LLVM/include")
set(CLANG_LIBRARY "$ENV{ProgramFiles}/LLVM/lib/libclang.lib")

set(SOURCES
    ${GAME_CONFIG_DIR}/Schema.cpp
    ${ENGINE_RUNTIME_DIR}/Runtime.cpp
    ${ENGINE_RUNTIME_DIR}/RuntimeObject.cpp
    ${CMAKE_SOURCE_DIR}/src/main.cpp
    ${CMAKE_SOURCE_DIR}/src/engine/RuntimeGen.cpp
    ${CMAKE_SOURCE_DIR}/include/model/store/ObjectStore.cpp
)

add_library(${PROJECT_NAME} SHARED ${SOURCES})

set(INCLUDES
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/src/config
    ${CMAKE_SOURCE_DIR}/src/core
    ${CMAKE_SOURCE_DIR}/src/engine
    ${CMAKE_SOURCE_DIR}/include/model
    ${CMAKE_SOURCE_DIR}/include/model/abstract
    ${CMAKE_SOURCE_DIR}/include/model/base
    ${CMAKE_SOURCE_DIR}/include/model/helpers
    ${CMAKE_SOURCE_DIR}/include/model/properties
    ${CMAKE_SOURCE_DIR}/include/model/resolve
    ${CMAKE_SOURCE_DIR}/include/model/store
    ${CMAKE_SOURCE_DIR}/include/model/types
    ${CMAKE_SOURCE_DIR}/src/schema
    ${DIALUP_INSTALL_DIR}/include/v1
    ${DIALUP_INSTALL_DIR}/include/external

    ${CMAKE_SOURCE_DIR}/config/rocketleague
    ${GAME_CONFIG_DIR}
    ${ENGINE_RUNTIME_DIR}
)
message(${GAME_CONFIG_DIR})

set(EXTERNAL_INCLUDES
    ${DIALUP_INSTALL_DIR}/include/external/fkYAML/include
    ${DIALUP_INSTALL_DIR}/include/external/fmt/include
)

target_include_directories(${PROJECT_NAME}
    PRIVATE ${CLANG_INCLUDE_DIR}
    ${INCLUDES}
    ${EXTERNAL_INCLUDES}
)

target_link_libraries(${PROJECT_NAME} PRIVATE ${CLANG_LIBRARY})
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$ENV{ProgramFiles}/LLVM/bin/libclang.dll"
        $<TARGET_FILE_DIR:${PROJECT_NAME}>
    COMMENT "Copying libclang.dll to output directory"
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    WINDOWS_EXPORT_ALL_SYMBOLS OFF
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    LOG_DIR="${LOG_DIR}"
    CONFIG_DIR="${CONFIG_DIR}"
    SDK_OUTPUT_DIR="${SDK_OUTPUT_DIR}"
    GAME_CONFIG_DIR="${GAME_CONFIG_DIR}"
    GAME_NAME="${GAME}"
    ENGINE_RUNTIME_DIR="${ENGINE_RUNTIME_DIR}"
    UE_MODEL_DIR="${UE_MODEL_DIR}"
    DIALUP_DIR="${DIALUP_DIR}"
)

target_compile_definitions(${PROJECT_NAME} PRIVATE SDK_BUILD)
target_compile_definitions(${PROJECT_NAME} PRIVATE _CRT_SECURE_NO_WARNINGS)

include("${DIALUP_INSTALL_DIR}/build-tools/PostProject.cmake")
